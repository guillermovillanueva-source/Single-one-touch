function renderCards() { const rawQuery = document.getElementById('searchInput').value.trim(); const query = normalize(rawQuery); const grid = document.getElementById('formsGrid'); const empty = document.getElementById('emptyState'); const filtered = services.filter(s => { // Excepción Bombavista en CO y CL if ((selectedCountry === 'CO' || selectedCountry === 'CL') && selectedBrand === 'bv') { return false; } // 1. Coincidencia de PAÍS: // Coincide si no hay país seleccionado O si el servicio es específico para ese país O si el servicio aplica a todos ('ALL') const matchCountry = !selectedCountry || s.country === selectedCountry || s.country === 'ALL'; // 2. Coincidencia de MARCA: // Coincide si no hay marca seleccionada O si la tarjeta es 'all' O si coincide con la marca seleccionada const matchBrand = !selectedBrand || s.brand === 'all' || s.brand === selectedBrand; // 3. Coincidencia de ÁREA / BU: // Coincide si no hay área seleccionada O si coincide exactamente con el área mapeada ('tiendas', 'ops', 'hq') let matchArea = !selectedArea || s.area === selectedArea; // Excepción existente para HQ en México if (selectedCountry === 'MX' && selectedArea === 'hq' && s.area === 'hq') { matchArea = true; } // 4. Búsqueda por texto const matchQuery = normalize(s.title).includes(query) || normalize(s.description).includes(query); return matchCountry && matchBrand && matchArea && matchQuery; }).sort((a, b) => (b.suggested ? 1 : 0) - (a.suggested ? 1 : 0)); let summaryText = []; if (selectedCountry) summaryText.push(COUNTRY_NAMES[selectedCountry]); if (selectedBrand) summaryText.push(BRAND_NAMES[selectedBrand]); if (selectedBU) summaryText.push(`BU: ${selectedBU}`); if (query) summaryText.push(`"${rawQuery}"`); document.getElementById('filterSummary').textContent = summaryText.length > 0 ? summaryText.join(' + ') : 'Todos los formularios'; document.getElementById('totalCards').textContent = `${filtered.length} Resultados`; if (filtered.length === 0) { grid.innerHTML = ''; empty.classList.remove('hidden'); return; } empty.classList.add('hidden'); grid.innerHTML = filtered.map(s => { let brandTag = ''; if (s.brand === 'bf') brandTag = 'BEN & FRANK'; else if (s.brand === 'bv') brandTag = 'bombavista'; const suggestedTag = s.suggested ? 'Sugerido' : ''; const actions = s.url ? `
` : `
`; return `
${s.flag} ${brandTag} ${suggestedTag} ${AREA_NAMES[s.area] || s.area}

${s.title}

${s.description}

${actions}
`}).join(''); }
